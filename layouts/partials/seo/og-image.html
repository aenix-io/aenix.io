{{- /* Social preview image of a page (og:image / twitter:image), as a site-relative path. Order: 1. images: in front matter (hand-picked or named card from scripts/generate-og-cards.py) 2. a per-page card from scripts/generate-og-cards.py, static/img/og/pages/.jpg, where is the page path with "/" replaced by "--" 3. cover_image: (blog posts whose card has not been generated yet) 4. the site-wide card (also the front page's) A blog post's cover_image stays its cover on the site, in the RSS feed and in its JSON-LD; only the link preview uses the brand card. */ -}} {{- $img := "img/aenix-social-card.jpg" -}} {{- with .Params.images -}} {{- $img = index . 0 -}} {{- else -}} {{- $key := trim .RelPermalink "/" | replaceRE "/" "--" -}} {{- $card := printf "img/og/pages/%s.jpg" $key -}} {{- if and (ne $key "") (fileExists (printf "static/%s" $card)) -}} {{- $img = $card -}} {{- else if .Params.cover_image -}} {{- $img = .Params.cover_image -}} {{- end -}} {{- end -}} {{- return (strings.TrimPrefix "/" $img) -}}

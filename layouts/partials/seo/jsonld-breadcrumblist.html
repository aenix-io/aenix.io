{{/* JSON-LD BreadcrumbList — every page. Auto-derived from URL structure unless .Params.breadcrumb is set in frontmatter (override). Frontmatter override shape: breadcrumb: - name: Home url: / - name: Solutions url: /solutions/ - name: DORA Compliance url: /solutions/dora-compliance/ Pages whose body already embeds a BreadcrumbList — the TCO calculator pages are synced from aenix-org/calculators with their own JSON-LD — set `skip_breadcrumb_jsonld: true` so the page does not ship two conflicting breadcrumb trails. See the cascade in hugo.yaml. */}} {{- if not .Params.skip_breadcrumb_jsonld -}} {{- $crumbs := slice -}} {{- if .Params.breadcrumb -}} {{- $crumbs = .Params.breadcrumb -}} {{- else -}} {{- /* Auto-derive from URL */ -}} {{- $segments := split (strings.TrimPrefix "/" (strings.TrimSuffix "/" .RelPermalink)) "/" -}} {{- $cumulative := "" -}} {{- /* German pages start at the German home page, not at the English one */ -}} {{- if and (gt (len $segments) 1) (eq (index $segments 0) "de") -}} {{- $crumbs = $crumbs | append (dict "name" "Startseite" "url" "/de/") -}} {{- $segments = after 1 $segments -}} {{- $cumulative = "/de" -}} {{- else -}} {{- $crumbs = $crumbs | append (dict "name" "Home" "url" "/") -}} {{- end -}} {{- $last := index $segments (sub (len $segments) 1) -}} {{- range $segments -}} {{- if . -}} {{- $cumulative = printf "%s/%s" $cumulative . -}} {{- if eq . $last -}} {{- /* the leaf reads better as the page's own title than as a humanized slug */ -}} {{- $crumbs = $crumbs | append (dict "name" ($.Title | default (. | humanize | title)) "url" (printf "%s/" $cumulative)) -}} {{- else -}} {{- /* only segments that are real pages: a blog post's /blog/2026/09/ has none and is a 404, and humanize made "2026th" and "9th" of them */ -}} {{- $seg := site.GetPage $cumulative -}} {{- if $seg -}} {{- $crumbs = $crumbs | append (dict "name" $seg.LinkTitle "url" $seg.RelPermalink) -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}}

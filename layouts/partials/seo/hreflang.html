{{/* seo/hreflang.html — Emits tags. Resolves locale pairs from frontmatter: - EN page declares hreflang_de: /de// and/or hreflang_ru: /ru// - DE page declares hreflang_en: // - RU page declares hreflang_en: // A pair is emitted only when the counterpart page exists and declares this page back (seo/locale-counterparts.html). One-way declarations — a DE URL that pairs with a different EN page, or a counterpart that was never built — are dropped instead of sending search engines a broken pair. If a counterpart exists, emits the pair plus `x-default` (always pointing at the EN version per Google's recommendation for sites with no neutral landing). If no counterpart is verified, an EN page emits only `x-default = self` (so single-language pages still pass hreflang validation). Called from seo/head.html. */}} {{- $c := partial "seo/locale-counterparts.html" . -}} {{- $lang := $c.lang -}} {{- $self := .Permalink -}} {{- $base := .Site.BaseURL | strings.TrimSuffix "/" -}} {{- if ne $lang "en" -}} {{/* Non-EN page (de/ru/fr) — emit self + EN counterpart (if verified) + x-default → EN */}} {{- with $c.en }} {{- else }} {{/* Non-EN page with no EN counterpart — emit no x-default at all. x-default names the version for users whose language we do not match; a German page is not that, and claiming it while an English site exists tells Google the wrong thing. Omitting it is valid. */}} {{- end }} {{- else -}} {{/* EN page (default) — emit EN self + verified counterparts + x-default → EN */}} {{- with $c.de }} {{- end }} {{- with $c.ru }} {{- end }} {{- end -}}

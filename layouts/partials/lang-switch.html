{{/* lang-switch.html — language toggle pill for pages that declare a locale counterpart in frontmatter (hreflang_en / hreflang_de / hreflang_ru). Rendered by _default/list.html and _default/event-landing.html in the hero. When it shows: - the EN↔RU pair (workshop pages), as before; - the EN↔DE pair, but only on event pages (layout or page_type `event-landing`, or under the tour / workshops / webinars sections and their /de/ twins, hubs included). DE landing pages and their EN counterparts keep their existing behavior (no visible switcher). Once shown, it lists every counterpart the page declares, in EN · DE · RU order, with the current language marked active. The DE locale is content under /de/ flagged with language: "de". */}} {{- $lang := .Params.language | default "en" -}} {{- if hasPrefix .RelPermalink "/de/" }}{{ $lang = "de" }}{{ end -}} {{- $ru := .Params.hreflang_ru -}} {{- $en := .Params.hreflang_en -}} {{- $de := .Params.hreflang_de -}} {{- $event := or (eq .Layout "event-landing") (eq (.Params.page_type | default "") "event-landing") -}} {{- range slice "/tour-2026/" "/workshops/" "/webinars/" "/de/tour-2026/" "/de/workshops/" "/de/webinare/" -}} {{- if hasPrefix $.RelPermalink . }}{{ $event = true }}{{ end -}} {{- end -}} {{- $ruPair := or (and (eq $lang "en") $ru) (and (eq $lang "ru") $en) -}} {{- $dePair := and $event (or (and (eq $lang "en") $de) (and (eq $lang "de") $en)) -}} {{- if $ruPair -}}
{{- if eq $lang "ru" }} English Русский {{- else }} English {{- if and $de $event }} Deutsch {{- end }} Русский {{- end }}
{{- else if $dePair -}}
{{- if eq $lang "de" }} English Deutsch {{- with $ru }} Русский {{- end }} {{- else }} English Deutsch {{- end }}
{{- end -}}
